    # 길이로 판단
    def check_result(self, segments):
        if len(segments) == 2:

            else:

    # 도로 boundary와 떨어진 거리에 따라 판단
    def check_result2(self, segments, wk_line) :
        if len(segments) == 2:

            l2:LineString = segments[1]
            
            intersect_roads = self.rd_gdf[self.rd_gdf['geometry'].intersects(wk_line)]

            mid_point_l1 = Point(l1.interpolate(0.5, normalized=True))

            distance_l2 = mid_point_l2.distance(road_boundary)
            
            if distance_l1 > distance_l2 : 
                return [l1]
            else :
                return [l2]
            
        return None


    # 두 벡터 사이의 각도 계산
    def angle_between_vectors(self, v1, v2):
        dot_product = np.dot(v1, v2)
        norms = np.linalg.norm(v1) * np.linalg.norm(v2)
        cos_theta = dot_product / norms
        cos_theta = np.clip(cos_theta, -1.0, 1.0)
        angle = np.arccos(cos_theta)
        return np.degrees(angle)

    # 두 라인이 닿는 점에서 각도가 90도 이상인 경우 병합
    def merge_lines_if_angles_allow(self, line1, line2):
        # 맞닫는 점을 찾기
        coords1 = list(line1.coords)
        coords2 = list(line2.coords)

        # 마지막 점의 인덱스
        last_point1 = coords1[-1]
        last_point2 = coords2[0]
        
        # 맞닫는 점과 끝 점이 서로 같은 경우
        if last_point1 == last_point2:
            angle = self.calculate_angle(coords1[-2], last_point1, coords2[1])
            if angle >= 95:
                # 병합
                return [LineString(coords1 + coords2[1:])]
            
        last_point1 = coords1[0]
        last_point2 = coords2[-1]
        
        # 맞닫는 점이란 끝 점이 서로 같은 경우
        if last_point1 == last_point2:
            angle = self.calculate_angle(coords1[-2], last_point1, coords2[1])
            if angle is np.isnan(angle).any() or angle >= 95:
                # 병합
                return [LineString(coords2 + coords1[1:])]
            
        return [line1, line2]
    
    # 세 점 사이의 각도 계산
    def calculate_angle(self, pointA, pointB, pointC):
        # 벡터 BA, BC 계산
        BA = np.array(pointA) - np.array(pointB)
        BC = np.array(pointC) - np.array(pointB)
        
        # 벡터 간의 내적 계산
        dot_product = np.dot(BA, BC)
        
        if dot_product  == 0:
            return 180
        
        # 벡터의 크기 계산
        norm_BA = np.linalg.norm(BA)
        norm_BC = np.linalg.norm(BC)
        
        # 각도 계산 (라디안)
        cos_theta = dot_product / (norm_BA * norm_BC)
        cos_theta = np.clip(cos_theta, -1.0, 1.0)  # 오차 방지를 위해 값 클리핑
        angle_radians = np.arccos(cos_theta)
        
        # 라디안을 도(degree)로 변환
        angle_degrees = np.degrees(angle_radians)
        
        return angle_degrees

    # 라인 객체 병합
    def merge_linestrings(self, line_strings):
        if not line_strings:
            return None  # 빈 리스트 처리

        merged_coords = []

        for line in line_strings:
            merged_coords.extend(list(line.coords))  # 모든 좌표를 추가

        return LineString(merged_coords)
